# 2. DOCKER MANAGEMENT & SUPERVISION

# Build and Start Containers in Background

# View Live Aggregated Logs
docker compose logs -f --tail=100

# View Live Logs for Monolith Application Container Only
docker compose logs -f algo_trading_monolith

# Restart the Monolith Container
docker compose restart algo_trading_monolith

# Stop and Remove All Containers
docker compose down

# Enter Interactive Bash Shell Inside Monolith Container
docker compose exec algo_trading_monolith bash

# Check Supervisor Process Status Inside Container (Web, WS Engine, Algo Engine)
docker compose exec algo_trading_monolith supervisorctl status

# Restart All Supervisor Sub-Services Inside Container
docker compose exec algo_trading_monolith supervisorctl restart all

# Restart Only the Algo Engine Inside Container
docker compose exec algo_trading_monolith supervisorctl restart algo_engine

# Restart Only the WebSocket Streaming Engine Inside Container
docker compose exec algo_trading_monolith supervisorctl restart websocket_engine

# Tail Supervisor Worker Engine Logs
docker compose exec algo_trading_monolith tail -f /app/logs/algo_engine.log
